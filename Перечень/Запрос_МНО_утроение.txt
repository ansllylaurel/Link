ВЫБРАТЬ
	Остатки.СчетДоверительногоУправления КАК СчетДоверительногоУправления,
	Остатки.Субконто1 КАК МНО,
	Настройки.УчетныйАктив КАК УчетныйАктив,
	Остатки.СуммаОстаток КАК Сумма
ИЗ
	РегистрБухгалтерии.БухгалтерскийУчет.Остатки(
			&Дата,
			Счет = ЗНАЧЕНИЕ(ПланСчетов.Основной.ПроцентыПоМинимальномуНеснижаемомуОстатку),
			,
			Организация = &Организация) КАК Остатки
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ РегистрСведений.НастройкиУчетаАктивов.СрезПоследних(
				&Дата,
				Организация = &Организация
					И УчетныйАктив В (ЗНАЧЕНИЕ(Справочник.УчетныеАктивы.ПроцентыНаМНО), ЗНАЧЕНИЕ(Справочник.УчетныеАктивы.ПроцентыНаСМО), ЗНАЧЕНИЕ(Справочник.УчетныеАктивы.ПроцентыНаФЕО))) КАК Настройки
		ПО Остатки.Организация = Настройки.Организация
			И Остатки.СчетДоверительногоУправления = Настройки.СчетДоверительногоУправления
			И Остатки.Счет = Настройки.СчетУчета
ГДЕ
	ТИПЗНАЧЕНИЯ(Остатки.Субконто1) = ТИП(Документ.МНО)
	И Остатки.СуммаОстаток <> 0
;
