// Формирует отчёт сверки с реестром. Читает xlsx, загружает данные 1С по пайщикам, выводит только расхождения.
// Параметры: Организация, ПолноеИмяФайла, ДокументРезультат (табличный документ), ДатаРеестра (если Неопределено - берётся из файла или не используется), ПИФ (если Неопределено - из файла).
// Возврат: Истина при успехе, иначе Ложь; текст ошибки в ДокументРезультат при ошибке.
Функция СформироватьОтчет(Организация, ПолноеИмяФайла, ДокументРезультат, ДатаРеестра = Неопределено, ПИФ = Неопределено) Экспорт
	ДокументРезультат.Очистить();
	Расширение = ВРег(СтрЗаменить(РаботаСФайламиКлиентСервер.ПолучитьРасширениеИмениФайла(ПолноеИмяФайла), ".", ""));
	Если Расширение <> "XLSX" Тогда
		ДокументРезультат.Область("R1C1").Текст = "Ошибка: допустим только формат xlsx. Указан: " + Расширение;
		Возврат Ложь;
	КонецЕсли;
	ТаблицаФайла = Новый ТабличныйДокумент;
	Попытка
		ТаблицаФайла.Прочитать(ПолноеИмяФайла, СпособЧтенияЗначенийТабличногоДокумента.Текст, ТипФайлаТабличногоДокумента.XLSX);
	Исключение
		ДокументРезультат.Область("R1C1").Текст = "Ошибка чтения файла: " + ОписаниеОшибки();
		Возврат Ложь;
	КонецПопытки;
	ДанныеИзФайла = ПрочитатьДанныеРеестраИзТабличногоДокумента(ТаблицаФайла, ДатаРеестра, ПИФ);
	Если ТипЗнч(ДанныеИзФайла) = Тип("Строка") Тогда
		ДокументРезультат.Область("R1C1").Текст = ДанныеИзФайла;
		Возврат Ложь;
	КонецЕсли;
	Если Не ЗначениеЗаполнено(ДанныеИзФайла.ПИФ) Или Не ЗначениеЗаполнено(ДанныеИзФайла.ДатаРеестра) Тогда
		ДокументРезультат.Область("R1C1").Текст = "В файле не определены дата реестра и/или ПИФ. Укажите их на форме или приведите файл к ожидаемому формату.";
		Возврат Ложь;
	КонецЕсли;
	Данные1С = ПолучитьДанныеПайщиковИз1С(Организация, ДанныеИзФайла.ПИФ, ДанныеИзФайла.ДатаРеестра);
	Расхождения = СформироватьТаблицуРасхождений(ДанныеИзФайла.Таблица, ДанныеИзФайла.ДатаРеестра, ДанныеИзФайла.ПИФ, Данные1С);
	ВывестиРасхожденияВТабличныйДокумент(Расхождения, ДокументРезультат, Организация, ДанныеИзФайла.ПИФ, ДанныеИзФайла.ДатаРеестра);
	Возврат Истина;
КонецФункции

Функция ПрочитатьДанныеРеестраИзТабличногоДокумента(ТаблицаФайла, ДатаРеестра, ПИФ)
	// Первая строка — заголовки. Ищем колонки по смыслу (подстрока в заголовке).
	НомерСтрокиЗаголовков = 1;
	МаксКол = ТаблицаФайла.ВысотаТаблицы;
	Если МаксКол = 0 Тогда
		Возврат "В файле нет данных.";
	КонецЕсли;
	МаксСтр = 0;
	Для НомерСтроки = 1 По ТаблицаФайла.ВысотаТаблицы Цикл
		Для НомерКолонки = 1 По ТаблицаФайла.ШиринаТаблицы Цикл
			Область = ТаблицаФайла.Область(НомерСтроки, НомерКолонки, НомерСтроки, НомерКолонки);
			Если Область.Объединение Или ЗначениеЗаполнено(СокрЛП(Область.Текст)) Тогда
				МаксСтр = Максимум(МаксСтр, НомерСтроки);
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;
	Ширина = ТаблицаФайла.ШиринаТаблицы;
	СтруктураКолонок = Новый Структура;
	Для НомерКолонки = 1 По Ширина Цикл
		Заголовок = СокрЛП(ТаблицаФайла.Область(НомерСтрокиЗаголовков, НомерКолонки, НомерСтрокиЗаголовков, НомерКолонки).Текст);
		ЗаголовокВерх = ВРег(Заголовок);
		Если СтрНайти(ЗаголовокВерх, "ФИО") > 0 Или (СтрНайти(ЗаголовокВерх, "ФАМИЛИЯ") > 0 Или СтрНайти(ЗаголовокВерх, "ИМЯ") > 0) Тогда
			СтруктураКолонок.Вставить("ФИО", НомерКолонки);
		ИначеЕсли СтрНайти(ЗаголовокВерх, "СЕРИЯ") > 0 И СтрНайти(ЗаголовокВерх, "ПАСПОРТ") = 0 И СтрНайти(ЗаголовокВерх, "УДОСТОВЕРЕН") = 0 Тогда
			СтруктураКолонок.Вставить("СерияПаспорта", НомерКолонки);
		ИначеЕсли (СтрНайти(ЗаголовокВерх, "НОМЕР") > 0 И СтрНайти(ЗаголовокВерх, "ПАСПОРТ") > 0) Или (СтрНайти(ЗаголовокВерх, "НОМЕР") > 0 И НомерКолонки > 1) Тогда
			Если Не СтруктураКолонок.Свойство("НомерПаспорта") Тогда
				СтруктураКолонок.Вставить("НомерПаспорта", НомерКолонки);
			КонецЕсли;
		ИначеЕсли СтрНайти(ЗаголовокВерх, "АДРЕС") > 0 Тогда
			СтруктураКолонок.Вставить("Адрес", НомерКолонки);
		ИначеЕсли СтрНайти(ЗаголовокВерх, "Р/С") > 0 Или СтрНайти(ЗаголовокВерх, "РАСЧЕТНЫЙ") > 0 Или (СтрНайти(ЗаголовокВерх, "СЧЕТ") > 0 И СтрНайти(ЗаголовокВерх, "НРД") = 0) Тогда
			СтруктураКолонок.Вставить("РасчетныйСчет", НомерКолонки);
		ИначеЕсли СтрНайти(ЗаголовокВерх, "НРД") > 0 Или СтрНайти(ЗаголовокВерх, "СЧЕТ НРД") > 0 Тогда
			СтруктураКолонок.Вставить("СчетНРД", НомерКолонки);
		ИначеЕсли СтрНайти(ЗаголовокВерх, "КОЛИЧЕСТВО") > 0 Или СтрНайти(ЗаголовокВерх, "ПАИ") > 0 Или СтрНайти(ЗаголовокВерх, "ДОЛЯ") > 0 Тогда
			СтруктураКолонок.Вставить("Количество", НомерКолонки);
		ИначеЕсли СтрНайти(ЗаголовокВерх, "ДАТА") > 0 И ДатаРеестра = Неопределено Тогда
			// не используем как колонку данных
		ИначеЕсли СтрНайти(ЗаголовокВерх, "ПИФ") > 0 И Не ЗначениеЗаполнено(ПИФ) Тогда
			// не используем как колонку данных
		КонецЕсли;
	КонецЦикла;
	Если Не СтруктураКолонок.Свойство("ФИО") Тогда
		Возврат "В файле не найдена колонка с ФИО (заголовок должен содержать ""ФИО"" или ""Фамилия""/""Имя"").";
	КонецЕсли;
	Таблица = Новый ТаблицаЗначений;
	Таблица.Колонки.Добавить("ФИО", Новый ОписаниеТипов("Строка", Новый КвалификаторыСтроки(500)));
	Таблица.Колонки.Добавить("СерияПаспорта", Новый ОписаниеТипов("Строка", Новый КвалификаторыСтроки(20)));
	Таблица.Колонки.Добавить("НомерПаспорта", Новый ОписаниеТипов("Строка", Новый КвалификаторыСтроки(20)));
	Таблица.Колонки.Добавить("Адрес", Новый ОписаниеТипов("Строка", Новый КвалификаторыСтроки(1000)));
	Таблица.Колонки.Добавить("РасчетныйСчет", Новый ОписаниеТипов("Строка", Новый КвалификаторыСтроки(50)));
	Таблица.Колонки.Добавить("СчетНРД", Новый ОписаниеТипов("Строка", Новый КвалификаторыСтроки(50)));
	Таблица.Колонки.Добавить("Количество", Новый ОписаниеТипов("Число", Новый КвалификаторыЧисла(18, 6)));
	Для НомерСтроки = 2 По МаксСтр Цикл
		ФИО = "";
		Если СтруктураКолонок.Свойство("ФИО") Тогда
			ФИО = СокрЛП(ТаблицаФайла.Область(НомерСтроки, СтруктураКолонок.ФИО, НомерСтроки, СтруктураКолонок.ФИО).Текст);
		КонецЕсли;
		Если НЕ ЗначениеЗаполнено(ФИО) Тогда
			Продолжить;
		КонецЕсли;
		НоваяСтрока = Таблица.Добавить();
		НоваяСтрока.ФИО = ФИО;
		НоваяСтрока.СерияПаспорта = СтруктураКолонок.Свойство("СерияПаспорта") ? СокрЛП(ТаблицаФайла.Область(НомерСтроки, СтруктураКолонок.СерияПаспорта, НомерСтроки, СтруктураКолонок.СерияПаспорта).Текст) : "";
		НоваяСтрока.НомерПаспорта = СтруктураКолонок.Свойство("НомерПаспорта") ? СокрЛП(ТаблицаФайла.Область(НомерСтроки, СтруктураКолонок.НомерПаспорта, НомерСтроки, СтруктураКолонок.НомерПаспорта).Текст) : "";
		НоваяСтрока.Адрес = СтруктураКолонок.Свойство("Адрес") ? СокрЛП(ТаблицаФайла.Область(НомерСтроки, СтруктураКолонок.Адрес, НомерСтроки, СтруктураКолонок.Адрес).Текст) : "";
		НоваяСтрока.РасчетныйСчет = СтруктураКолонок.Свойство("РасчетныйСчет") ? СокрЛП(ТаблицаФайла.Область(НомерСтроки, СтруктураКолонок.РасчетныйСчет, НомерСтроки, СтруктураКолонок.РасчетныйСчет).Текст) : "";
		НоваяСтрока.СчетНРД = СтруктураКолонок.Свойство("СчетНРД") ? СокрЛП(ТаблицаФайла.Область(НомерСтроки, СтруктураКолонок.СчетНРД, НомерСтроки, СтруктураКолонок.СчетНРД).Текст) : "";
		ЗначКол = 0;
		Если СтруктураКолонок.Свойство("Количество") Тогда
			ТекстКол = СокрЛП(ТаблицаФайла.Область(НомерСтроки, СтруктураКолонок.Количество, НомерСтроки, СтруктураКолонок.Количество).Текст);
			Если ЗначениеЗаполнено(ТекстКол) Тогда
				ЗначКол = Число(СтрЗаменить(ТекстКол, " ", ""));
			КонецЕсли;
		КонецЕсли;
		НоваяСтрока.Количество = ЗначКол;
	КонецЦикла;
	// Попытка прочитать дату и ПИФ из первых ячеек, если не переданы
	Если ДатаРеестра = Неопределено Или Не ЗначениеЗаполнено(ДатаРеестра) Тогда
		Для НомерКолонки = 1 По Минимум(5, Ширина) Цикл
			ТекстЯчейки = СокрЛП(ТаблицаФайла.Область(1, НомерКолонки, 1, НомерКолонки).Текст);
			Если СтрНайти(ВРег(ТекстЯчейки), "ДАТА") > 0 И НомерКолонки < Ширина Тогда
				ЗначДаты = СокрЛП(ТаблицаФайла.Область(2, НомерКолонки, 2, НомерКолонки).Текст);
				Если ЗначениеЗаполнено(ЗначДаты) Тогда
					Попытка
						ДатаРеестра = XMLЗначение(Тип("Дата"), ЗначДаты);
					Исключение
						ДатаРеестра = Неопределено;
					КонецПопытки;
				КонецЕсли;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	Если ПИФ = Неопределено Или Не ЗначениеЗаполнено(ПИФ) Тогда
		Для НомерКолонки = 1 По Минимум(10, Ширина) Цикл
			ТекстЯчейки = СокрЛП(ТаблицаФайла.Область(1, НомерКолонки, 1, НомерКолонки).Текст);
			Если СтрНайти(ВРег(ТекстЯчейки), "ПИФ") > 0 И НомерКолонки <= Ширина Тогда
				ЗначПИФ = СокрЛП(ТаблицаФайла.Область(2, НомерКолонки, 2, НомерКолонки).Текст);
				Если ЗначениеЗаполнено(ЗначПИФ) Тогда
					ПИФ = Справочники.ПаевыеИнвестиционныеФонды.НайтиПоНаименованию(ЗначПИФ, Истина);
					Если Не ЗначениеЗаполнено(ПИФ) Тогда
						ПИФ = Справочники.ПаевыеИнвестиционныеФонды.НайтиПоКоду(ЗначПИФ, Истина);
					КонецЕсли;
				КонецЕсли;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	Результат = Новый Структура("Таблица, ДатаРеестра, ПИФ", Таблица, ?(ДатаРеестра = Неопределено, Дата(1,1,1), ДатаРеестра), ПИФ);
	Возврат Результат;

Функция СформироватьТаблицуРасхождений(ТаблицаФайла, ДатаРеестра, ПИФ, Данные1С)
	// Сверка по ФИО (нормализованная строка). Счёт НРД — по общему количеству.
	Расхождения = Новый ТаблицаЗначений;
	Расхождения.Колонки.Добавить("ФИО", Новый ОписаниеТипов("Строка"));
	Расхождения.Колонки.Добавить("ВидРасхождения", Новый ОписаниеТипов("Строка"));
	Расхождения.Колонки.Добавить("Поле", Новый ОписаниеТипов("Строка"));
	Расхождения.Колонки.Добавить("ВФайле", Новый ОписаниеТипов("Строка"));
	Расхождения.Колонки.Добавить("В1С", Новый ОписаниеТипов("Строка"));
	Словарь1С = Новый Соответствие;
	Для Каждого Строка1С Из Данные1С Цикл
		ФИО1С = СокрЛП(Строка1С.Фамилия + " " + Строка1С.Имя + " " + Строка1С.Отчество);
		Словарь1С.Вставить(ВРег(ФИО1С), Строка1С);
	КонецЦикла;
	СуммаКоличестваВФайле = 0;
	СуммаКоличестваВ1С = 0;
	Для Каждого СтрокаФайла Из ТаблицаФайла Цикл
		СуммаКоличестваВФайле = СуммаКоличестваВФайле + СтрокаФайла.Количество;
	КонецЦикла;
	Для Каждого Строка1С Из Данные1С Цикл
		СуммаКоличестваВ1С = СуммаКоличестваВ1С + Строка1С.Количество;
	КонецЦикла;
	Если СуммаКоличестваВФайле <> СуммаКоличестваВ1С Тогда
		Новая = Расхождения.Добавить();
		Новая.ФИО = "(итого по реестру)";
		Новая.ВидРасхождения = "Счет НРД (общее количество)";
		Новая.Поле = "Количество";
		Новая.ВФайле = Формат(СуммаКоличестваВФайле, "ЧГ=0");
		Новая.В1С = Формат(СуммаКоличестваВ1С, "ЧГ=0");
	КонецЕсли;
	Для Каждого СтрокаФайла Из ТаблицаФайла Цикл
		КлючФИО = ВРег(СокрЛП(СтрокаФайла.ФИО));
		Строка1С = Словарь1С.Получить(КлючФИО);
		Если Строка1С = Неопределено Тогда
			Новая = Расхождения.Добавить();
			Новая.ФИО = СтрокаФайла.ФИО;
			Новая.ВидРасхождения = "Нет в 1С";
			Новая.Поле = "—";
			Новая.ВФайле = "Есть в файле";
			Новая.В1С = "";
			Продолжить;
		КонецЕсли;
		ПаспортФайл = СокрЛП(СтрокаФайла.СерияПаспорта + " " + СтрокаФайла.НомерПаспорта);
		Паспорт1С = СокрЛП(Строка1С.СерияПаспорта + " " + Строка1С.НомерПаспорта);
		Если ПаспортФайл <> Паспорт1С И (ЗначениеЗаполнено(ПаспортФайл) Или ЗначениеЗаполнено(Паспорт1С)) Тогда
			Новая = Расхождения.Добавить();
			Новая.ФИО = СтрокаФайла.ФИО;
			Новая.ВидРасхождения = "Паспорт";
			Новая.Поле = "Паспорт";
			Новая.ВФайле = ПаспортФайл;
			Новая.В1С = Паспорт1С;
		КонецЕсли;
		АдресФайл = СокрЛП(СтрокаФайла.Адрес);
		Адрес1С = СокрЛП(Строка1С.Адрес);
		Если АдресФайл <> Адрес1С И (ЗначениеЗаполнено(АдресФайл) Или ЗначениеЗаполнено(Адрес1С)) Тогда
			Новая = Расхождения.Добавить();
			Новая.ФИО = СтрокаФайла.ФИО;
			Новая.ВидРасхождения = "Адрес";
			Новая.Поле = "Адрес";
			Новая.ВФайле = АдресФайл;
			Новая.В1С = Адрес1С;
		КонецЕсли;
		РсФайл = СокрЛП(СтрокаФайла.РасчетныйСчет);
		Рс1С = СокрЛП(Строка1С.НомерСчета);
		Если РсФайл <> Рс1С И (ЗначениеЗаполнено(РсФайл) Или ЗначениеЗаполнено(Рс1С)) Тогда
			Новая = Расхождения.Добавить();
			Новая.ФИО = СтрокаФайла.ФИО;
			Новая.ВидРасхождения = "Расчетный счет";
			Новая.Поле = "Р/с";
			Новая.ВФайле = РсФайл;
			Новая.В1С = Рс1С;
		КонецЕсли;
	КонецЦикла;
	// Пайщики, которые есть в 1С, но нет в файле
	Для Каждого Строка1С Из Данные1С Цикл
		ФИО1С = СокрЛП(Строка1С.Фамилия + " " + Строка1С.Имя + " " + Строка1С.Отчество);
		НайденВФайле = Ложь;
		Для Каждого СтрокаФайла Из ТаблицаФайла Цикл
			Если ВРег(СокрЛП(СтрокаФайла.ФИО)) = ВРег(ФИО1С) Тогда
				НайденВФайле = Истина;
				Прервать;
			КонецЕсли;
		КонецЦикла;
		Если Не НайденВФайле Тогда
			Новая = Расхождения.Добавить();
			Новая.ФИО = ФИО1С;
			Новая.ВидРасхождения = "Нет в файле";
			Новая.Поле = "—";
			Новая.ВФайле = "";
			Новая.В1С = "Есть в 1С";
		КонецЕсли;
	КонецЦикла;
	Возврат Расхождения;
КонецФункции

ШАГ 2 — Обработка ИмпортКотировокММВБ
  • ЗАМЕНИТЬ вызов: вместо 
      ОтчетОРасхождениях = ПодготовитьОтчетОРасхождениях(Расхождения, Измерения, РесурсыИРеквизиты);
    использовать:
      ОтчетОРасхождениях = РаботаСОтчетами.ПодготовитьОтчетОРасхождениях(Расхождения, Измерения, РесурсыИРеквизиты,
        Новый Структура("НетРасхождений, НоваяЗапись, ЗаписьУдалена, ЗаписьИзменена", мТипыЗаписейВОтчетеОРасхождениях_НетРасхождений, мТипыЗаписейВОтчетеОРасхождениях_НоваяЗапись, мТипыЗаписейВОтчетеОРасхождениях_ЗаписьУдалена, мТипыЗаписейВОтчетеОРасхождениях_ЗаписьИзменена),
        ЭтотОбъект.СкрыватьВОтчетеНовыеКотировки);
  • УДАЛИТЬ ЦЕЛИКОМ функцию ПодготовитьОтчетОРасхождениях из модуля обработки (от строки с "Функция ПодготовитьОтчетОРасхождениях" до "КонецФункции" включительно).

ШАГ 3 — Дополнительный отчёт СверкаСРеестромВладельцевИнвестиционныхПаев
  • Создать новую внешнюю обработку (Внешняя обработка / Дополнительный отчёт). Сохранить как epf.
  • В модуль объекта вставить ЦЕЛИКОМ код из блока "МОДУЛЬ ОБЪЕКТА" выше.
  • Создать форму с реквизитами и элементами по описанию. В модуль формы вставить код из блока "МОДУЛЬ ФОРМЫ".
  • Зарегистрировать обработку в справочнике ДополнительныеОтчетыИОбработки (загрузить epf).

ШАГ 4 — Проверка
  • Убедиться, что обработка ИмпортКотировокММВБ по-прежнему формирует отчёт о расхождениях по котировкам.
  • Запустить новый отчёт, указать организацию и файл xlsx, сформировать — проверить вывод только расхождений.
  • При несовпадении структуры xlsx с ожидаемой (заголовки колонок) — донастроить маппинг в функции ПрочитатьДанныеРеестраИзТабличногоДокумента по фактическим заголовкам в 1.xlsx / 2.xlsx.
