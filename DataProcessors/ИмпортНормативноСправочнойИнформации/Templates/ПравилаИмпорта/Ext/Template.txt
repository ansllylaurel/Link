SELECT
	CAST(Organizations.OrganizationID AS varchar) [Идентификатор],
	ReferenceBooks.dbo.RemoveExtraSpaces([Organizations].ShortOrganizationName) [Наименование],

	CAST([Bankrupts].StateBeginDate AS datetime) [ДатаБанкротства]
FROM
	ReferenceBooks.dbo.Organizations [Organizations]
	LEFT JOIN
		(SELECT DISTINCT
			CASE
				WHEN TableName = 'Bonds' THEN [Bonds].IssuerID
				WHEN TableName = 'Stocks' THEN [Stocks].IssuerID
				WHEN TableName = 'Shares' THEN [Shares].IssuerID
			END OrganizationID,
			StateBeginDate
		FROM
			ReferenceBooks.dbo.StatesEntries [StatesEntries]
			LEFT JOIN [Migration].[AM].[vw_Bonds] [Bonds] ON [StatesEntries].BondID = [Bonds].BondID
			LEFT JOIN ReferenceBooks.dbo.Stocks [Stocks] ON [StatesEntries].StockID = [Stocks].StockID
			LEFT JOIN ReferenceBooks.dbo.Shares [Shares] ON [StatesEntries].ShareID = [Shares].ShareID
		WHERE
			StateID = 9	-- Эмитент-банкрот
		) [Bankrupts]
		ON
			[Bankrupts].OrganizationID = [Organizations].OrganizationID
WHERE
	[Organizations].OrganizationID IN (&amp;ИдентификаторыВыбранныхОбъектов) AND Organizations.ShortOrganizationName IS NOT NULL

SELECT
	CAST(IsNUll(MAX(DATEADD (day, 1, SubDocuments.Date_Expiration)), Documents.Date_Issue) AS datetime) [Период],
	'ПолучитьОбъектПоКодуВнешнейСистемы("' + CAST(Documents.ID AS varchar) + '", "Лицензии", ИСТИНА)' [$$Лицензия],
	'ПолучитьОбъектПоКодуВнешнейСистемы("' + CAST(Documents.OrganizationID AS varchar) + '", "ЮридическиеЛица")' [$$ЮрФизЛицо],
	'ПолучитьОбъектПоКодуВнешнейСистемы("' + CAST(Documents.Sub_Type_UID AS varchar(100)) + '", "ВидыДеятельности")' [$$ВидДеятельности],
	'Перечисления.СтатусыДокументов.Действует' [$Статус],
	2 [Порядок]
FROM
 	[Migration].[AM].[vw_Licenses] AS Documents
	LEFT JOIN [Migration].[AM].[vw_Licenses] AS SubDocuments
		ON Documents.OrganizationID = SubDocuments.OrganizationID
			AND Documents.Sub_Type_UID = SubDocuments.Sub_Type_UID
			AND NOT Documents.ID = SubDocuments.ID
			AND (Documents.Date_Expiration > SubDocuments.Date_Expiration
				OR (Documents.Date_Expiration IS NULL AND NOT SubDocuments.Date_Expiration IS NULL))
WHERE
	(Documents.status_Type_UID = 'Document.Status.Active' OR NOT Documents.Date_Expiration IS NULL)
	AND Documents.OrganizationID IN (&amp;ИдентификаторыВыбранныхОбъектов) 
 
GROUP BY
	Documents.Date_Issue,
	Documents.Date_Expiration,
	Documents.ID,
	Documents.OrganizationID,
	Documents.Sub_Type_UID
	
UNION ALL

	Case
		WHEN Status_Type_UID = 'Document.Status.Recalled' THEN 'Перечисления.СтатусыДокументов.Отозван'
		WHEN Status_Type_UID = 'Document.Status.Cancelled' THEN 'Перечисления.СтатусыДокументов.Аннулирован'

  FROM

	AND NOT status_Type_UID = 'Document.Status.Active'
	AND OrganizationID IN (&amp;ИдентификаторыВыбранныхОбъектов)

WITH cte AS(
	SELECT
		T.Date_From,
		T.Date_To,
		T.Firm_ID,
		T.Parent_Firm_ID,
		T.Firm_ID AS Child
	FROM
		ReferenceBooks.dbo.Firms_Firms T
	WHERE
		T.RelationType = 'Firms.Parent.Holding'
		AND T.Firm_ID IN(&amp;ИдентификаторыВыбранныхОбъектов)
	
	UNION ALL
	
	SELECT
		T.Date_From,
		T.Date_To,
		T.Firm_ID,
		T.Parent_Firm_ID,
		cte.Child
	FROM
		ReferenceBooks.dbo.Firms_Firms T
		JOIN cte
			ON T.Firm_ID = cte.Parent_Firm_ID AND T.RelationType = 'Firms.Parent.Holding'
)
SELECT 
	CAST(cte.Date_From AS datetime) AS [НачалоПериода],
	CAST(cte.Date_To AS datetime) AS [КонецПериода],
	'ПолучитьОбъектПоКодуВнешнейСистемы("' + CAST(cte.Child AS varchar) + '", "ЮридическиеЛица", Истина)' AS [$$ЮридическоеЛицо],
	'ПолучитьОбъектПоКодуВнешнейСистемы("' + CAST(cte.Parent_Firm_ID AS varchar) + '", "ЮридическиеЛица", Истина)' AS [$$РодительскоеЮридическоеЛицо],
	'Справочники.ТипыСвязейЮрФизЛиц.ГруппаСвязанныхЛиц' AS [$ТипСвязи]
FROM
	cte cte
	LEFT JOIN ReferenceBooks.dbo.Firms_Firms AS T
	ON cte.Parent_Firm_ID = T.Firm_ID AND T.RelationType = 'Firms.Parent.Holding'
WHERE
	T.Firm_ID IS NULL

UNION ALL
