=====================================================================
0. Цель изменений
=====================================================================
Нужно реализовать полный поток из ТЗ:
1) сформировать XML по стратегиям;
2) сохранить XML в сетевую папку периода;
3) сверить новый XML с предыдущим файлом стратегии;
4) при расхождениях отправить отдельное уведомление;
5) при отсутствии расхождений разместить на сайте (HTTP);
6) отправить итоговое письмо с результатами и ссылками;
7) вести журнал действий.

---------------------------------------------------------------------
Важно до старта
---------------------------------------------------------------------
1. В примерах ниже используется путь:
   \\g1\shares\UK\Client\РегионЭсМ\ОТДЕЛ КОНТРОЛЯ\Раскрытие информации\ИИС\Доходности стандартный стратегии
   и подпапка yyyy-MM.
2. Лучше вынести путь/URL/токен в настройки, но для быстрого старта
   допускается временно захардкодить (потом вынести).
3. Все новые блоки Исключение должны быть непустыми (лог + ВызватьИсключение
   или корректная обработка).

		КонецЦикла;

Функция СравнитьXMLФайлы(ПутьТекущегоФайла, ПутьПредыдущегоФайла) Экспорт

	Расхождения = Новый Массив;

	Если ПутьПредыдущегоФайла = "" Тогда
		Возврат Расхождения;
	КонецЕсли;

	Текущий = Новый ТекстовыйДокумент;
	Текущий.Прочитать(ПутьТекущегоФайла);

	Предыдущий = Новый ТекстовыйДокумент;
	Предыдущий.Прочитать(ПутьПредыдущегоФайла);

	СтрокиТекущий = СтрРазделить(Текущий.ПолучитьТекст(), Символы.ПС, Ложь);
	СтрокиПредыдущий = СтрРазделить(Предыдущий.ПолучитьТекст(), Символы.ПС, Ложь);

	МаксСтрок = ?(СтрокиТекущий.Количество() > СтрокиПредыдущий.Количество(), СтрокиТекущий.Количество(), СтрокиПредыдущий.Количество());

	Для Индекс = 0 По МаксСтрок - 1 Цикл

		СтрокаТекущий = ?(Индекс < СтрокиТекущий.Количество(), СтрокиТекущий[Индекс], "");
		СтрокаПредыдущий = ?(Индекс < СтрокиПредыдущий.Количество(), СтрокиПредыдущий[Индекс], "");

		Если СтрокаТекущий <> СтрокаПредыдущий Тогда
			Расхождения.Добавить(
				"Строка " + Строка(Индекс + 1) + ": " + СтрокаПредыдущий + " => " + СтрокаТекущий);
		КонецЕсли;

	КонецЦикла;

	Возврат Расхождения;

КонецФункции

	РезультатСверки = Новый ТаблицаЗначений;
	РезультатСверки.Колонки.Добавить("ИмяФайла");
	РезультатСверки.Колонки.Добавить("ЕстьРасхождения", Новый ОписаниеТипов("Булево"));
	РезультатСверки.Колонки.Добавить("ОписаниеРасхождений");

	Для Каждого СтрокаСохранения Из РезультатСохранения Цикл

		СтрокаСверки = РезультатСверки.Добавить();
		СтрокаСверки.ИмяФайла = СтрокаСохранения.ИмяФайла;
		СтрокаСверки.ЕстьРасхождения = Ложь;
		СтрокаСверки.ОписаниеРасхождений = "";

		Если Не СтрокаСохранения.Успешно Тогда
			Продолжить;
		КонецЕсли;

		ПутьПредыдущегоФайла = НайтиПутьПредыдущегоФайлаСтратегии(СтрокаСохранения.ИмяФайла, Период);
		Расхождения = СравнитьXMLФайлы(СтрокаСохранения.ПолныйПуть, ПутьПредыдущегоФайла);

		Если Расхождения.Количество() > 0 Тогда
			СтрокаСверки.ЕстьРасхождения = Истина;
			СтрокаСверки.ОписаниеРасхождений = СтрСоединить(Расхождения, Символы.ПС);
		КонецЕсли;

	КонецЦикла;

	Возврат РезультатСверки;

Процедура ВыполнитьРазмещениеДоходностиНаСайте(ПолучателиОповещения) Экспорт

	// 1) Готовим пакет данных так же, как в ОтправитьОтчет()
	// Рекомендуется вынести блок "Запрос/Пакет" из ОтправитьОтчет в отдельную функцию ПолучитьПакетДляДоходности().
	Пакет = ПолучитьПакетДляДоходности();

	// 2) Формируем XML
	Вложения = СформироватьXMLВложенияПоСтратегиям(Пакет);

	// 3) Сохраняем в сеть
	КаталогПериода = ПолучитьКаталогПериода(ЭтотОбъект.Период);
	РезультатСохранения = СохранитьXMLФайлыВСеть(Вложения, КаталогПериода);

	// 4) Сверяем с предыдущим периодом
	РезультатСверки = ПроверитьРасхожденияПоСохраненнымФайлам(РезультатСохранения, ЭтотОбъект.Период);

	// 5) Размещаем те файлы, где нет расхождений и сохранение успешно
	РезультатРазмещения = Новый ТаблицаЗначений;
	РезультатРазмещения.Колонки.Добавить("ИмяФайла");
	РезультатРазмещения.Колонки.Добавить("Ссылка");
	РезультатРазмещения.Колонки.Добавить("Успешно", Новый ОписаниеТипов("Булево"));
	РезультатРазмещения.Колонки.Добавить("ТекстОшибки");

	Для Каждого СтрокаСохранения Из РезультатСохранения Цикл

		СтрокаРазмещения = РезультатРазмещения.Добавить();
		СтрокаРазмещения.ИмяФайла = СтрокаСохранения.ИмяФайла;
		СтрокаРазмещения.Успешно = Ложь;

		Если Не СтрокаСохранения.Успешно Тогда
			СтрокаРазмещения.ТекстОшибки = "Не сохранен файл в сеть";
			Продолжить;
		КонецЕсли;

		СтрокаСверки = РезультатСверки.Найти(СтрокаСохранения.ИмяФайла, "ИмяФайла");
		Если СтрокаСверки <> Неопределено И СтрокаСверки.ЕстьРасхождения Тогда
			СтрокаРазмещения.ТекстОшибки = "Обнаружены расхождения с предыдущим периодом";
			Продолжить;
		КонецЕсли;

		РезультатHTTP = РазместитьФайлНаСайте(СтрокаСохранения.ПолныйПуть, СтрокаСохранения.ИмяФайла);
		СтрокаРазмещения.Успешно = РезультатHTTP.Успешно;
		СтрокаРазмещения.Ссылка = РезультатHTTP.Ссылка;
		СтрокаРазмещения.ТекстОшибки = РезультатHTTP.ТекстОшибки;

	КонецЦикла;

	// 6) Уведомления: о расхождениях и итоговое о размещении
	ОтправитьОповещениеОРасхождениях(ПолучателиОповещения, РезультатСверки);

9. Добавить итоговое письмо по размещению (ссылки/предупреждения)
=====================================================================
Файл: Ext\ObjectModule.bsl

--- НОВЫЙ КОД ---
Процедура ОтправитьИтоговыйОтчетОРазмещении(ПолучателиОповещения, РезультатРазмещения, РезультатСверки) Экспорт

	Текст = "Информация о размещении данных на сайте " + Формат(ТекущаяДата(), "ДФ=dd.MM.yyyy") + Символы.ПС + Символы.ПС;
	Номер = 0;

	Для Каждого СтрокаРазмещения Из РезультатРазмещения Цикл
		Номер = Номер + 1;
		Текст = Текст + Строка(Номер) + ". " + СтрокаРазмещения.ИмяФайла + Символы.ПС;

		Если СтрокаРазмещения.Успешно Тогда
			Текст = Текст + "Ссылка: " + СтрокаРазмещения.Ссылка + Символы.ПС;
		Иначе
			Текст = Текст + "Внимание! Информация о доходности стратегии не размещена. "
				+ СтрокаРазмещения.ТекстОшибки + Символы.ПС;
		КонецЕсли;

		Текст = Текст + Символы.ПС;
	КонецЦикла;

	Тема = "РЕГИОН ЭсМ. Размещение финансовых показателей по ИИС на "
		+ Формат(ТекущаяДата(), "ДФ=dd.MM.yy") + " на сайте";

	ПараметрыПисьма = Новый Структура(
		"Кому, Тема, Тело, ТипТекста",
		ПолучателиОповещения,
		Тема,
		Текст,
		ТипТекстаПочтовогоСообщения.Текст);

	РаботаСЭлектроннойПочтой.ОтправитьЭлектронноеПисьмо(
		РаботаСЭлектроннойПочтой.ПолучитьСистемнуюУчетнуюЗапись(),
		ПараметрыПисьма);
