SELECT DISTINCT
	Account,
	CAST(DealNumber AS VARCHAR) DealNumber,
	DealDate + CONVERT(DATETIME, DealTime) DealDate,
	RepaymentDate,
	CONVERT(DATETIME, '1/1/1 ' + DealTime) DealTime,
	InstrumentShortName,
	InstrumentName,
	CASE WHEN InstrumentClassCode LIKE '%FUT%' THEN P_InstrumentName ELSE InstrumentCode END InstrumentCode,
	InstrumentClass,
	InstrumentClassCode,
	Operation,
	CASE WHEN InstrumentClassCode LIKE '%FUT%' THEN Volume/Quantity ELSE Price END Price,
	Quantity,
	Volume,
	IsNULL(PaymentCurrency, Currency) AS Currency,
	PaymentCode,
	Coupon,
	CASE WHEN CHARINDEX('/', Client) = 0 THEN Client ELSE LEFT(Client, CHARINDEX('/', Client) - 1) END Client,
	REPLACE(Client, '/', '_') Briefcase,
	PartnerOrganization,
	IsNULL(TSComission, 0) Comission,
	P_LotSize
FROM
	[#SourceSchema].[Deals] D (NOLOCK)
	LEFT JOIN [#SourceSchema].[Parameters] P (NOLOCK)
		ON P.P_TradeDate = D.DealDate AND P.P_InstrumentClassCode = D.InstrumentClassCode AND P.P_InstrumentCode = D.InstrumentCode
WHERE
	D.DealDate = @DealDate
	AND D.Dealer IN (@Dealer, 'SPBFUT')
	AND NOT D.InstrumentClassCode IN ('GPB_OTCFX', 'CETS', 'TRAN', 'RFND')
	AND (CASE WHEN CHARINDEX('/', Client) = 0 THEN Client ELSE LEFT(Client, CHARINDEX('/', Client) - 1) END IN (SELECT Code FROM #ClientsByOrganization)
			OR REPLACE(Client, '/', '_') IN (SELECT Code FROM #ClientsByOrganization))
	</ПравилоИмпорта>

	<ПравилоИмпорта ГруппаОбъектов="СделкиРЕПО" ВидДокумента="СделкаРЕПО">
IF OBJECT_ID('[#SourceSchema].[DealsREPO]') IS NOT NULL AND OBJECTPROPERTY(OBJECT_ID('[#SourceSchema].[DealsREPO]'), 'IsTable') = 1
	(SELECT DISTINCT
		CASE WHEN D.DealType LIKE 'Вторая часть%' THEN 2 ELSE 1 END RepoPart,
		CASE WHEN D.DealType LIKE 'Вторая часть%' THEN CAST(D1.DealNumber AS VARCHAR) ELSE NULL END FirstPart,
		CAST(D.OrderlNumber AS VARCHAR) OrderNumber,
		CAST(D.DealNumber AS VARCHAR) DealNumber,
		CAST(D.SourceNumber AS VARCHAR) SourceNumber,
		CASE WHEN D.DealType LIKE 'Вторая часть%' THEN D.RepaymentDate ELSE D.DealDate + CONVERT(DATETIME, D.DealTime) END DealDate,
		CASE WHEN D.DealType LIKE 'Вторая часть%' THEN CONVERT(DATETIME, '1/1/1') ELSE D.RepaymentDate END RepaymentDate,
		CASE WHEN D.DealType LIKE 'Вторая часть%' THEN CONVERT(DATETIME, '1/1/1') ELSE CONVERT(DATETIME, '1/1/1 ' + D.DealTime) END DealTime,
		D.InstrumentShortName,
		D.InstrumentName,
		D.InstrumentCode,
		D.InstrumentClass,
		D.InstrumentClassCode,
		D.Operation,
		D.Price,
		D.Quantity,
		D.Volume,
		CASE WHEN IsNULL(P.P_LotSize, 0) = 0 THEN 0
				WHEN D.DealType LIKE 'Вторая часть%' THEN ROUND(CAST(D1.RepoVolume - D1.RepoYield AS decimal(28, 12)) / (D1.Quantity * P.P_LotSize), 10, 0)  
				ELSE ROUND(CAST(D.Volume AS decimal(28, 12)) / (D.Quantity * P.P_LotSize), 6, 0)
		END Цена,
		CASE 	
			WHEN D.DealType LIKE 'Вторая часть%' THEN 0
			ELSE ROUND((D.RepoSum - D.Coupon) / (D.Quantity * P.P_LotSize), 6, 0)
		END ЦенаРепо_М, 
		D.Coupon,
		CASE WHEN IsNULL(P.P_LotSize, 0) = 0 THEN 0 ELSE ROUND(CAST(D.Coupon AS decimal(28, 14)) / (D.Quantity * P.P_LotSize), 12) END ЦенаКупона, 
		D.PriceReturn,
		D.RepoYield,
		D.RepoSum,
		D.RepoVolume,
		CASE 	
			WHEN D.DealType LIKE 'Вторая часть%' THEN 0
			WHEN IsNULL(P.P_LotSize, 0) = 0 THEN 0
			ELSE ROUND(CAST(D.RepoVolume - D.RepoYield AS decimal(28, 12)) / (D.Quantity * P.P_LotSize) , 10, 0) 
		END  ЦенаВторойЧасти,
		CASE 	
			WHEN D.DealType LIKE 'Вторая часть%' THEN 0
			WHEN IsNULL(P.P_LotSize, 0) = 0 THEN 0
			ELSE ROUND(CAST(D.RepoYield AS decimal(28, 14)) / (D.Quantity * P.P_LotSize), 12) 
		END  ЦенаКупонаВторойЧасти,
		D.Duration,
		D.RepoRate,
		IsNULL(D.PaymentCurrency, D.Currency) AS Currency,
		CASE WHEN CHARINDEX('/', D.Client) = 0 THEN D.Client ELSE LEFT(D.Client, CHARINDEX('/', D.Client) - 1) END Client,
		REPLACE(D.Client, '/', '_') Briefcase,
		D.Account,
		D.PartnerOrganization,
		IsNULL(D.TSComission, 0) Comission,
		P.P_LotSize,
		D.AccountCode,
		D.RepoStartingStartDiscount  
	FROM
		[#SourceSchema].[DealsREPO] D (NOLOCK)
		LEFT JOIN [#SourceSchema].[DealsREPO] S (NOLOCK) -- (source) консолидирующие сделки для РЕПО с ЦК
			ON S.DealNumber = D.SourceNumber
		LEFT JOIN [#SourceSchema].[DealsREPO] D1 (NOLOCK)
			ON D1.SourceNumber = S.DealNumber AND D1.DealType LIKE 'Первая часть%'
		LEFT JOIN [#SourceSchema].[Parameters] P (NOLOCK)
			ON P.P_TradeDate = D.DealDate AND P.P_InstrumentClassCode = D.InstrumentClassCode AND P.P_InstrumentCode = D.InstrumentCode
	WHERE
		D.DealDate = @DealDate
			AND D.Dealer = @Dealer
			AND NOT D.Operation = 'К/П' -- для загрузки сделки с ЦК необходимо чтобы в таблице присутствовало три записи (1-ая ч., 2-ая ч. и консолидирующая часть "К/П"), если в таблице присутствует только консолидирующая часть - этих данных не достаточно для загрузки, исключаем из загрузки
			AND NOT D.DealNumber IN ( -- искл. консолидирующую запись по РЕПО с ЦК
				SELECT IsNULL(SourceNumber, 0)
				FROM [#SourceSchema].[DealsREPO] (NOLOCK)
				WHERE DealDate = @DealDate AND Dealer = @Dealer)
			AND (CASE WHEN CHARINDEX('/', D.Client) = 0 THEN D.Client ELSE LEFT(D.Client, CHARINDEX('/', D.Client) - 1) END IN (SELECT Code FROM #ClientsByOrganization)
					OR REPLACE(D.Client, '/', '_') IN (SELECT Code FROM #ClientsByOrganization))
		
	UNION ALL
	
	SELECT DISTINCT
		2 AS RepoPart,
		CAST(D2.FirstPathNumber AS VARCHAR) AS FirstPart,
		CAST(D2.OrderlNumber AS VARCHAR) AS OrderNumber,
		CAST(D2.FirstPathNumber AS VARCHAR) AS DealNumber,
		'',
		D2.RepaymentDate,
		CONVERT(DATETIME, '1/1/1'),
		CONVERT(DATETIME, '1/1/1'),
		D1.InstrumentShortName,
		D1.InstrumentName,
		D1.InstrumentCode,
		D1.InstrumentClass,
		D1.InstrumentClassCode,
		D2.Operation,
		D1.PriceReturn,
		D1.Quantity,
		D1.RepoVolume - D1.RepoYield,
		CASE WHEN IsNULL(P_LotSize, 0) = 0 THEN 0 ELSE ROUND(CAST((D1.RepoVolume - D1.RepoYield) AS decimal(28, 12)) / (D1.Quantity * P_LotSize), 10, 0) END,
		0, 
		D1.RepoYield,
		CASE WHEN IsNULL(P_LotSize, 0) = 0 THEN 0 ELSE ROUND(CAST(D1.RepoYield AS decimal(28, 14)) / (D1.Quantity * P_LotSize), 12) END,
		0,
		D1.RepoYield,
		0,
		D1.RepoVolume,
		0,
		0,
		0,
		D2.RepoRate,
		IsNULL(D1.PaymentCurrency, D1.Currency),
		CASE WHEN CHARINDEX('/', D2.Client) = 0 THEN D2.Client ELSE LEFT(D2.Client, CHARINDEX('/', D2.Client) - 1) END,
		REPLACE(D2.Client, '/', '_'),
		D2.Account,
		D2.PartnerOrganization,
		IsNULL(D2.TSComission, 0),
		P_LotSize,
		D2.AccountCode,
		D1.RepoStartingStartDiscount
	FROM
		[#SourceSchema].[DealsREPOCl] D2 (NOLOCK)
		LEFT JOIN [#SourceSchema].[DealsREPO] D1 (NOLOCK)
			ON D1.DealNumber = D2.FirstPathNumber
		LEFT JOIN [#SourceSchema].[Parameters] P (NOLOCK)
			ON P.P_TradeDate = D1.DealDate AND P.P_InstrumentClassCode = D1.InstrumentClassCode AND P.P_InstrumentCode = D1.InstrumentCode
		
	WHERE
		D2.DealDate = @DealDate
			AND D2.Dealer = @Dealer
			AND D2.Operation IN ('Купля', 'Продажа')
			AND (CASE WHEN CHARINDEX('/', D2.Client) = 0 THEN D2.Client ELSE LEFT(D2.Client, CHARINDEX('/', D2.Client) - 1) END IN (SELECT Code FROM #ClientsByOrganization)
					OR REPLACE(D2.Client, '/', '_') IN (SELECT Code FROM #ClientsByOrganization))
	)
ELSE
	(SELECT NULL WHERE 1 = 0)
	</ПравилоИмпорта>
	
	<ПравилоИмпорта ГруппаОбъектов="СделкиСВалютой" ВидДокумента="СделкаСВалютой">
SELECT DISTINCT
	D.Account,
	CAST(D.DealNumber AS VARCHAR) DealNumber,
	D.DealDate + CONVERT(DATETIME, D.DealTime) DealDate,
	D.RepaymentDate,
	CONVERT(DATETIME, '1/1/1 ' + D.DealTime) DealTime,
	D.InstrumentCode,
	D.InstrumentClassCode,
	D.Operation,
	D.Price,
	D.Quantity,
	D.Volume,
	LEFT(D.InstrumentShortName, 3) AS Currency,
	SUBSTRING(D.InstrumentShortName, 4, 3) AS CounterCurrency,
	D.PaymentCode,
	CASE
		WHEN CHARINDEX('/', D.Client) > 0 THEN LEFT(D.Client, CHARINDEX('/', D.Client) - 1)
		WHEN CHARINDEX('V', D.Client) > 0 THEN LEFT(D.Client, CHARINDEX('V', D.Client) - 1)
		ELSE D.Client
	END Client,
	REPLACE(D.Client, '/', '_') Briefcase,
	D.PartnerOrganization,
	IsNULL(D.TSComission, 0) Comission,
	P.P_LotSize
FROM
	[#SourceSchema].[Deals] D (NOLOCK)
	LEFT JOIN [#SourceSchema].[Parameters] P (NOLOCK)
		ON P.P_TradeDate = D.DealDate AND P.P_InstrumentClassCode = D.InstrumentClassCode AND P.P_InstrumentCode = D.InstrumentCode
WHERE
	D.DealDate = @DealDate
		--AND D.Dealer = @Dealer
		AND D.InstrumentClassCode IN ('CETS', 'SDBP', 'FUTS', 'FIXS', 'WAPS', 'CNGD', 'FUTN', 'FIXN', 'WAPN', 'CETS_SU')
		AND CASE
				WHEN CHARINDEX('/', D.Client) > 0 THEN LEFT(D.Client, CHARINDEX('/', D.Client) - 1)
				WHEN CHARINDEX('V', D.Client) > 0 THEN LEFT(D.Client, CHARINDEX('V', D.Client) - 1)
				ELSE D.Client
			END IN (SELECT Code FROM #ClientsByOrganization)
	</ПравилоИмпорта>
